# E2E Testing Setup Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Stand up a complete Playwright TypeScript E2E suite covering auth, real-time chat delivery (< 3s), multi-tab presence (AFK within 2s), file upload/download (5 MB), and admin ban enforcement; run the suite headless in Docker against the full Compose stack.

**Architecture:** Tests use isolated Playwright `BrowserContext` instances per user (separate cookie/localStorage namespaces) to simulate concurrent users without any shared client state. Users are created via REST API calls in fixture setup — no pre-seeded DB records required. The AFK test calls `SetAfk()` directly on the SignalR hub from `page.evaluate()` (bypassing the 60s inactivity timer) using a dev-only `window.__presenceHub` reference exposed by `PresenceService`. Test reports are written to a host-mounted volume for post-run inspection.

**Tech Stack:** Playwright 1.49, TypeScript 5.4, `@microsoft/signalr` 8.x (for direct hub validation in helpers), Node 22 Alpine, Docker Compose profiles (e2e service only runs with `--profile e2e`).

```typescript
import { APIRequestContext, request } from '@playwright/test';

export interface TestUser {
  id:           string;
  email:        string;
  password:     string;
  username:     string;
  accessToken:  string;
  refreshToken: string;
}

export interface RoomDto {
  id:   string;
  name: string;
}

export class ApiHelpers {
  private readonly baseUrl: string;

  constructor() {
    this.baseUrl = process.env.BASE_URL ?? 'http://localhost';
  }

  // ─── Auth ────────────────────────────────────────────────────────────────────

  async register(): Promise<TestUser> {
    const ctx = await this.context();
    const suffix   = `${Date.now()}-${Math.random().toString(36).slice(2, 6)}`;
    const email    = `e2e-${suffix}@test.local`;
    const password = 'Test@1234!E2E';
    const username = `user_${suffix}`;

    const reg = await ctx.post('/api/auth/register', {
      data: { email, password, username },
    });
    if (!reg.ok()) throw new Error(`register failed (${reg.status()}): ${await reg.text()}`);

    const login = await ctx.post('/api/auth/login', {
      data: { email, password },
    });
    if (!login.ok()) throw new Error(`login failed (${login.status()}): ${await login.text()}`);

    const body = await login.json();
    await ctx.dispose();

    return {
      id:           body.user.id,
      email,
      password,
      username,
      accessToken:  body.accessToken,
      refreshToken: body.refreshToken,
    };
  }

  // ─── Rooms ───────────────────────────────────────────────────────────────────

  async createRoom(
    accessToken: string,
    opts: { name?: string; isPublic?: boolean } = {},
  ): Promise<RoomDto> {
    const ctx = await this.authContext(accessToken);
    const res = await ctx.post('/api/rooms', {
      data: {
        name:     opts.name ?? `e2e-room-${Date.now()}`,
        topic:    'E2E test room',
        isPublic: opts.isPublic ?? true,
        maxMembers: 50,
      },
    });
    if (!res.ok()) throw new Error(`createRoom failed (${res.status()}): ${await res.text()}`);
    const room = await res.json();
    await ctx.dispose();
    return room;
  }

  async addMember(roomId: string, userId: string, adminToken: string): Promise<void> {
    const ctx = await this.authContext(adminToken);
    const res = await ctx.post(`/api/rooms/${roomId}/members/${userId}`);
    if (!res.ok()) throw new Error(`addMember failed (${res.status()}): ${await res.text()}`);
    await ctx.dispose();
  }

  async banMember(roomId: string, userId: string, adminToken: string): Promise<void> {
    const ctx = await this.authContext(adminToken);
    // DELETE /api/rooms/{roomId}/members/{userId} = room ban (AGENT.md §9 / §16)
    const res = await ctx.delete(`/api/rooms/${roomId}/members/${userId}`);
    if (!res.ok()) throw new Error(`banMember failed (${res.status()}): ${await res.text()}`);
    await ctx.dispose();
  }

  // ─── Internals ───────────────────────────────────────────────────────────────

  private async context(): Promise<APIRequestContext> {
    return request.newContext({ baseURL: this.baseUrl });
  }

  private async authContext(accessToken: string): Promise<APIRequestContext> {
    return request.newContext({
      baseURL: this.baseUrl,
      extraHTTPHeaders: { Authorization: `Bearer ${accessToken}` },
    });
  }
}
```

test.describe('Presence Engine', () => {

  test('User B appears Online to User A after joining', async ({
    userAPage, userBPage, userA, userB, api,

    await userBPage.goto(`/rooms/${room.id}`);

    // Wait for User A's member list to show User B as online.
    const statusDot = userAPage.locator(`[data-testid="member-status-${userB.id}"]`);
    await expect(statusDot).toHaveAttribute('data-status', 'online', { timeout: 3_000 });
  });

  test('User B AFK status propagates to User A within 2 seconds', async ({
    userAPage, userBPage, userA, userB, api,
  }) => {
    const room = await api.createRoom(userA.accessToken);
    await api.addMember(room.id, userB.id, userA.accessToken);

    await Promise.all([
      userAPage.goto(`/rooms/${room.id}`),
      userBPage.goto(`/rooms/${room.id}`),
    ]);

    const statusDot = userAPage.locator(`[data-testid="member-status-${userB.id}"]`);
    await expect(statusDot).toHaveAttribute('data-status', 'online', { timeout: 3_000 });

    // Directly invoke SetAfk() via the dev-mode hub reference.
    // This bypasses the 60s inactivity timer — makes the test deterministic and fast.
    await userBPage.evaluate(async () => {
      const hub = (window as any).__presenceHub;
      if (!hub) throw new Error('__presenceHub not set — PresenceService must expose it in isDevMode()');
      await hub.invoke('SetAfk');
    });

    // User A must see AFK within 2 seconds (SLA from AGENT.md §12).
    await expect(statusDot).toHaveAttribute('data-status', 'afk', { timeout: 2_000 });
  });

  test('User B returns to Online after SetActive', async ({
    userAPage, userBPage, userA, userB, api,
  }) => {
    const room = await api.createRoom(userA.accessToken);
    await api.addMember(room.id, userB.id, userA.accessToken);

    await Promise.all([
      userAPage.goto(`/rooms/${room.id}`),
      userBPage.goto(`/rooms/${room.id}`),
    ]);

    // Go AFK first.
    await userBPage.evaluate(async () => {
      await (window as any).__presenceHub?.invoke('SetAfk');
    });
    const statusDot = userAPage.locator(`[data-testid="member-status-${userB.id}"]`);
    await expect(statusDot).toHaveAttribute('data-status', 'afk', { timeout: 2_000 });

    // Come back active.
    await userBPage.evaluate(async () => {
      await (window as any).__presenceHub?.invoke('SetActive');
    });
    await expect(statusDot).toHaveAttribute('data-status', 'online', { timeout: 2_000 });
  });

  test('User B appears Offline to User A after closing tab', async ({
    userAPage, userBPage, userA, userB, api,
  }) => {
    const room = await api.createRoom(userA.accessToken);
    await api.addMember(room.id, userB.id, userA.accessToken);

    await Promise.all([
      userAPage.goto(`/rooms/${room.id}`),
      userBPage.goto(`/rooms/${room.id}`),
    ]);

    await userBPage.close();

    await expect(statusDot).toHaveAttribute('data-status', 'offline', { timeout: 3_000 });
  });
});

  test('upload 5 MB file and User B can download it', async ({
    userAPage, userBPage, userA, userB, api,
  }) => {
    const room = await api.createRoom(userA.accessToken);
    await api.addMember(room.id, userB.id, userA.accessToken);

    await Promise.all([
      userAPage.goto(`/rooms/${room.id}`),
      userBPage.goto(`/rooms/${room.id}`),
    ]);

    await expect(userAPage.locator('[data-testid="chat-area"]')).toBeVisible({ timeout: 5_000 });
    await expect(userBPage.locator('[data-testid="chat-area"]')).toBeVisible({ timeout: 5_000 });

    // Write a 5 MB binary temp file — filled with a recognisable pattern.
    const tmpPath = path.join(os.tmpdir(), `e2e-5mb-${Date.now()}.bin`);
    fs.writeFileSync(tmpPath, Buffer.alloc(5 * 1024 * 1024, 0xab));

    // Use setInputFiles — works even when the <input type="file"> is hidden.
    await userAPage.setInputFiles('[data-testid="file-input"]', tmpPath);
    await userAPage.click('[data-testid="upload-submit"]');

    // User B must see the attachment message within 15 seconds (upload + SignalR push).
    const downloadLink = userBPage.locator('[data-testid="attachment-download-link"]').first();
    await expect(downloadLink).toBeVisible({ timeout: 15_000 });

    // Verify the download endpoint returns 200 (membership-validated file access).
    const [response] = await Promise.all([
      userBPage.waitForResponse(
        (resp) => resp.url().includes('/api/files/') && resp.request().method() === 'GET',
        { timeout: 10_000 },
      ),
      downloadLink.click(),
    ]);
    expect(response.status()).toBe(200);

    // Cleanup temp file.
    fs.unlinkSync(tmpPath);
  });

  test('non-member cannot download attachment (403)', async ({
    userA, userB, api,
  }) => {
    // Create a room with only User A, upload a file, then try to download as User B.
    const room = await api.createRoom(userA.accessToken, { isPublic: false });

    // Use the API directly to upload without UI (simpler and faster).
    const { request } = await import('@playwright/test');
    const tmpPath = path.join(os.tmpdir(), `e2e-403-${Date.now()}.bin`);
    fs.writeFileSync(tmpPath, Buffer.alloc(1024, 0xff));

    const ctx = await request.newContext({
      baseURL: process.env.BASE_URL ?? 'http://localhost',
      extraHTTPHeaders: { Authorization: `Bearer ${userA.accessToken}` },
    });
    const uploadRes = await ctx.post('/api/files/upload', {
      multipart: {
        file: { name: 'test.bin', mimeType: 'application/octet-stream', buffer: fs.readFileSync(tmpPath) },
        roomId: room.id,
      },
    });
    expect(uploadRes.ok()).toBe(true);
    const { attachmentId } = await uploadRes.json();

    // User B (not a member) attempts to download — must get 403.
    const downloadRes = await request.newContext({
      baseURL: process.env.BASE_URL ?? 'http://localhost',
      extraHTTPHeaders: { Authorization: `Bearer ${userB.accessToken}` },
    });
    const forbiddenRes = await (await downloadRes).get(`/api/files/${attachmentId}`);
    expect(forbiddenRes.status()).toBe(403);

    await ctx.dispose();
    fs.unlinkSync(tmpPath);
  });
});
